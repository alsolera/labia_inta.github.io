% AeroML logo — TikZ version, for LaTeX papers, posters and slides.
% The website uses images/logo.svg; this file is the print equivalent.
% Keep the two in sync: the geometry below is copied from that file.
%
% Usage:
%   \usepackage{tikz}
%   \input{brand/logo-tikz.tex}
%   \aeromllogo          % 1 unit = 1pt, so about 42 mm wide
%   \aeromllogo[0.6]     % smaller
%
% The picture uses yscale=-1 so coordinates match the SVG viewBox directly
% (y grows downward); keep that in mind before editing a control point.
%
% The SVG fades the streamlines from navy to violet along x. A stroke gradient
% is not worth the trouble in TikZ, so here they are solid navy. Everything
% else is identical.

\definecolor{aeromlnavy}{HTML}{171B2B}
\definecolor{aeromlviolet}{HTML}{4B3F9E}
\definecolor{aeromlpaper}{HTML}{F9F8F6}

\newcommand{\aeromllogo}[1][1]{%
\begin{tikzpicture}[
    x=#1 pt, y=#1 pt, yscale=-1,
    line cap=round, line join=round,
    node mark/.style={circle, draw=aeromlviolet, fill=aeromlpaper,
                      line width=#1*0.6pt, inner sep=0pt,
                      minimum size=#1*6.6pt},
  ]

  % streamlines: one smooth curve each, into the three input nodes
  \draw[aeromlnavy, line width=#1*0.67pt]
    (2,9) .. controls (22,2.6) and (44,3.2) .. (62.6,7.6);
  \draw[aeromlnavy, line width=#1*0.67pt]
    (3,15) .. controls (22,9.2) and (44,12.4) .. (62.6,19.4);

  % third streamline, detached: starts clear of the trailing edge
  \draw[aeromlnavy, line width=#1*0.6pt]
    (56.6,33.2) .. controls (58.8,33.1) and (60.6,32.6) .. (62.8,31.9);

  % airfoil: round leading edge left, sharp trailing edge right
  \draw[aeromlnavy, line width=#1*1.06pt]
    (51,32.9) .. controls (41,26.9) and (30,20.8) .. (20.8,21.5)
              .. controls (13.4,22.1) and (7.3,25.8) .. (8.5,29.4)
              .. controls (9.5,32.8) and (18.5,34.4) .. (30,34.4)
              .. controls (40,34.4) and (47,33.5) .. (51,32.9) -- cycle;

  % generic feedforward network, 3 - 2 - 2, fully connected
  \foreach \sy in {8,20,32}{%
    \foreach \ty in {14,26}{%
      \draw[aeromlviolet, line width=#1*0.42pt] (66,\sy) -- (87,\ty);}}
  \foreach \sy in {14,26}{%
    \foreach \ty in {14,26}{%
      \draw[aeromlviolet, line width=#1*0.42pt] (87,\sy) -- (108,\ty);}}

  \foreach \y in {8,20,32}{\node[node mark] at (66,\y) {};}
  \foreach \y in {14,26}{\node[node mark] at (87,\y) {};}
  \foreach \y in {14,26}{\node[node mark] at (108,\y) {};}
\end{tikzpicture}%
}
